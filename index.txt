<script>
function updateDoc(){
  const val = id => document.getElementById(id).value;
  document.getElementById('out-name').textContent = val('petName') || 'Sin nombre';
  document.getElementById('out-sub').textContent = val('species') + ' · ' + (val('breed') || '—');
  document.getElementById('out-age').textContent = val('age') || '—';
  document.getElementById('out-weight').textContent = val('weight') || '—';
  document.getElementById('out-neutered').textContent = val('neutered');
  document.getElementById('out-vaccines').textContent = val('vaccines') === 'Al día' ? 'Vacunas al día' : 'Vacunación pendiente';
  document.getElementById('out-history').textContent = val('history') || '—';
  document.getElementById('out-vet').textContent = val('vet') || '—';
  document.getElementById('out-owner').textContent = val('ownerName') || '—';
  document.getElementById('out-city').textContent = val('city') || '—';

  const traitsList = document.getElementById('out-traits');
  traitsList.innerHTML = '';
  document.querySelectorAll('.trait:checked').forEach(cb=>{
    const li = document.createElement('li');
    li.textContent = cb.dataset.text;
    traitsList.appendChild(li);
  });
  if(!traitsList.children.length){
    const li = document.createElement('li');
    li.style.opacity = .6;
    li.textContent = 'Sin rasgos marcados todavía';
    li.style.listStyle = 'none';
    traitsList.appendChild(li);
  }
}
document.querySelectorAll('#tool1 input, #tool1 select, #tool1 textarea').forEach(el=>{
  el.addEventListener('input', updateDoc);
  el.addEventListener('change', updateDoc);
});
updateDoc();

document.querySelectorAll('.city-tab').forEach(tab=>{
  tab.addEventListener('click', ()=>{
    document.querySelectorAll('.city-tab').forEach(t=>t.setAttribute('aria-selected','false'));
    tab.setAttribute('aria-selected','true');
    document.getElementById('links-madrid').style.display = tab.dataset.city==='madrid' ? 'grid' : 'none';
    document.getElementById('links-barcelona').style.display = tab.dataset.city==='barcelona' ? 'grid' : 'none';
  });
});
</script>
